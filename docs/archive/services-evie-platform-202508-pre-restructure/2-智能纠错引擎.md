## 一、模块定位

智能纠错引擎是 ASR 和 AI Agent 之间的**语义增强层**。

```
ASR 原始文本  →  纠错引擎  →  企业标准语言
```

核心职责：将 ASR 产生的"不准确文本"转换为符合企业语境的标准化文本。

## 二、纠错流水线架构

```
ASR 文本输入
    │
    ▼
┌─────────────┐
│ 阶段1: 预处理 │  分词 → 拼音转换 → 实体识别
└──────┬──────┘
       ▼
┌─────────────┐
│ 阶段2: 召回  │  规则库 / 企业字典 / ES / Milvus / LLM
└──────┬──────┘
       ▼
┌─────────────┐
│ 阶段3: 多路  │  规则纠错 → 拼音匹配 → 编辑距离 → 实体匹配
│      纠错    │
└──────┬──────┘
       ▼
┌─────────────┐
│ 阶段4: 融合  │  加权评分 → 候选排序
│      评分    │
└──────┬──────┘
       ▼
┌─────────────┐
│ 阶段5: 决策  │  置信度 ≥0.95 → 自动修正
│             │  置信度 0.75-0.95 → 二次确认
│             │  置信度 <0.75 → 保留原文
└─────────────┘
```

## 三、核心 Go 接口

```go
// internal/biz/correction.go

// CorrectionEngine 纠错引擎接口
type CorrectionEngine interface {
    // Correct 执行纠错全流程
    Correct(ctx context.Context, req CorrectionRequest) (*CorrectionResult, error)
}

// Corrector 单路纠错器（策略模式）
type Corrector interface {
    // Name 纠错器名称
    Name() string
    // Correct 返回候选修正列表
    Correct(ctx context.Context, text string, tokens []Token) ([]CorrectionCandidate, error)
    // Priority 优先级，数字越大越先执行
    Priority() int
}

type CorrectionRequest struct {
    TenantID   string
    UserID     string
    SessionID  string
    Text       string   // ASR 原始文本
    Context    string   // 上下文（前几轮对话）
    Segments   []ASRSegment
}

type Token struct {
    Index   int
    Text    string
    Pinyin  string
    Pos     string // 词性
    Entity  string // 实体类型: person/org/product/time/quantity
}

type CorrectionCandidate struct {
    From       string
    To         string
    Type       string  // person/org/product/dictionary/rule
    Confidence float64
    Source     string  // rule/pinyin/entity/llm
}

type CorrectionResult struct {
    Original    string
    Corrected   string
    Changes     []CorrectionChange
    Confidence  float64
    NeedConfirm bool     // 是否需要用户确认
}

type CorrectionChange struct {
    From       string
    To         string
    Type       string
    Confidence float64
}
```

## 四、五阶段编排实现

```go
// internal/biz/correction.go

func (e *CorrectionEngineImpl) Correct(ctx context.Context, req CorrectionRequest) (*CorrectionResult, error) {
    // ===== 阶段1: 预处理 =====
    tokens := e.tokenize(req.Text)
    tokens = e.assignPinyin(tokens)
    tokens = e.detectEntities(ctx, req.TenantID, tokens)

    // ===== 阶段2+3: 多路召回与纠错（并发执行）=====
    var allCandidates []CorrectionCandidate
    candidateCh := make(chan []CorrectionCandidate, len(e.correctors))

    for _, c := range e.correctors {
        go func(cor Corrector) {
            candidates, _ := cor.Correct(ctx, req.Text, tokens)
            candidateCh <- candidates
        }(c)
    }

    for range e.correctors {
        candidates := <-candidateCh
        allCandidates = append(allCandidates, candidates...)
    }

    // ===== 阶段4: 融合评分 =====
    ranked := e.rankAndDeduplicate(allCandidates, tokens, req.Context)

    // ===== 阶段5: 决策输出 =====
    var changes []CorrectionChange
    var needConfirm bool
    output := req.Text

    for _, c := range ranked {
        if c.Confidence >= 0.95 {
            // 自动修正
            output = strings.Replace(output, c.From, c.To, 1)
            changes = append(changes, CorrectionChange{
                From: c.From, To: c.To,
                Type: c.Type, Confidence: c.Confidence,
            })
        } else if c.Confidence >= 0.75 {
            // 需要二次确认
            needConfirm = true
            changes = append(changes, CorrectionChange{
                From: c.From, To: c.To,
                Type: c.Type, Confidence: c.Confidence,
            })
        }
        // <0.75 直接丢弃
    }

    return &CorrectionResult{
        Original:    req.Text,
        Corrected:   output,
        Changes:     changes,
        Confidence:  e.overallConfidence(changes),
        NeedConfirm: needConfirm,
    }, nil
}
```

## 五、各路纠错器实现

### 5.1 规则纠错器（优先级 100，<1ms）

```go
// internal/biz/correction_rule.go

type RuleCorrector struct {
    rules map[string]string // source → target（按 tenant_id 加载）
}

func (r *RuleCorrector) Name() string    { return "rule" }
func (r *RuleCorrector) Priority() int   { return 100 }

func (r *RuleCorrector) Correct(ctx context.Context, text string, tokens []Token) ([]CorrectionCandidate, error) {
    var candidates []CorrectionCandidate
    for _, t := range tokens {
        if target, ok := r.rules[t.Text]; ok {
            candidates = append(candidates, CorrectionCandidate{
                From: t.Text, To: target,
                Type: "dictionary", Confidence: 1.0, Source: "rule",
            })
        }
    }
    return candidates, nil
}
```

### 5.2 拼音纠错器（优先级 80，<20ms）

```go
// internal/biz/correction_pinyin.go

type PinyinCorrector struct {
    dictCenter DictionaryCenter // 字典中心，提供拼音索引
}

func (p *PinyinCorrector) Name() string  { return "pinyin" }
func (p *PinyinCorrector) Priority() int { return 80 }

func (p *PinyinCorrector) Correct(ctx context.Context, text string, tokens []Token) ([]CorrectionCandidate, error) {
    var candidates []CorrectionCandidate
    for _, t := range tokens {
        // 跳过已标记为实体的 token
        if t.Entity != "" {
            continue
        }
        // 用拼音查字典中心的候选
        samePinyin := p.dictCenter.FindByPinyin(ctx, t.Pinyin)
        for _, entry := range samePinyin {
            if entry.Word == t.Text {
                continue
            }
            // 上下文评分（前后 token 的语义一致性）
            score := p.contextScore(tokens, t.Index, entry)
            if score > 0.6 {
                candidates = append(candidates, CorrectionCandidate{
                    From: t.Text, To: entry.Word,
                    Type: "pinyin", Confidence: score, Source: "pinyin",
                })
            }
        }
    }
    return candidates, nil
}
```

### 5.3 编辑距离纠错器（优先级 70，<20ms）

```go
// internal/biz/correction_editdistance.go

type EditDistanceCorrector struct {
    dictCenter DictionaryCenter
    maxDist    int // 默认 2
}

func (e *EditDistanceCorrector) Name() string  { return "edit_distance" }
func (e *EditDistanceCorrector) Priority() int { return 70 }

func (e *EditDistanceCorrector) Correct(ctx context.Context, text string, tokens []Token) ([]CorrectionCandidate, error) {
    var candidates []CorrectionCandidate
    for _, t := range tokens {
        if t.Entity != "" || len(t.Text) < 2 {
            continue
        }
        // 模糊搜索字典中编辑距离 <= maxDist 的词
        similar := e.dictCenter.FuzzySearch(ctx, t.Text, e.maxDist)
        for _, entry := range similar {
            dist := levenshteinDistance(t.Text, entry.Word)
            conf := 1.0 - float64(dist)/float64(max(len(t.Text), len(entry.Word)))
            if conf > 0.7 {
                candidates = append(candidates, CorrectionCandidate{
                    From: t.Text, To: entry.Word,
                    Type: "edit_distance", Confidence: conf, Source: "edit_distance",
                })
            }
        }
    }
    return candidates, nil
}
```

### 5.4 企业实体纠错器（优先级 90，<50ms）

```go
// internal/biz/correction_entity.go

type EntityCorrector struct {
    dictCenter DictionaryCenter
}

func (e *EntityCorrector) Name() string  { return "entity" }
func (e *EntityCorrector) Priority() int { return 90 }

func (e *EntityCorrector) Correct(ctx context.Context, text string, tokens []Token) ([]CorrectionCandidate, error) {
    var candidates []CorrectionCandidate
    for _, t := range tokens {
        // 尝试匹配实体：人员/组织/产品
        entry := e.dictCenter.MatchEntity(ctx, EntityMatchRequest{
            TenantID: "", // 从 ctx 提取
            Text:     t.Text,
            Type:     "", // null = 不限类型
        })
        if entry != nil {
            candidates = append(candidates, CorrectionCandidate{
                From:       t.Text,
                To:         entry.CanonicalName,
                Type:       entry.Type, // person/org/product
                Confidence: entry.Score,
                Source:     "entity",
            })
        }
    }
    return candidates, nil
}
```

## 六、评分模型

```go
// 加权评分公式: Score = 0.3*拼音相似 + 0.3*企业匹配 + 0.2*上下文 + 0.2*LLM
func (e *CorrectionEngineImpl) scoreCandidate(c CorrectionCandidate, tokens []Token, idx int) float64 {
    score := 0.0

    switch c.Source {
    case "rule":
        score = 1.0 // 规则命中直接满分
    case "entity":
        score = 0.4*c.Confidence + 0.3*c.ContextMatch + 0.2*c.PositionMatch + 0.1*c.HistoryMatch
    case "pinyin":
        score = 0.3*c.Confidence + 0.3*c.DictMatch + 0.2*c.ContextMatch + 0.2*c.EditDistanceScore
    case "edit_distance":
        score = c.Confidence * 0.7 // 仅依赖编辑距离，降权
    case "llm":
        score = c.Confidence * 0.8 // LLM 置信度通常偏高，微调
    }

    return math.Min(score, 1.0)
}
```

## 八、置信度决策

```go
type ConfidenceLevel int

const (
    HighConfidence   ConfidenceLevel = iota // ≥0.95 自动修正
    MediumConfidence                        // 0.75-0.95 二次确认
    LowConfidence                           // <0.75 保留原文
)

func (e *CorrectionEngineImpl) classifyConfidence(confidence float64) ConfidenceLevel {
    switch {
    case confidence >= 0.95:
        return HighConfidence
    case confidence >= 0.75:
        return MediumConfidence
    default:
        return LowConfidence
    }
}
```

func (e *CorrectionEngineImpl) requestConfirmation(ctx context.Context, changes []CorrectionChange) error {
    for _, ch := range changes {
        // 调用通知中心发送确认消息

            Metadata: map[string]string{
                "from": ch.From,
                "to":   ch.To,
                "type": ch.Type,
            },
        })
    }
    return nil
}

## 九、用户反馈闭环

```go
// 记录用户反馈，形成企业个性化学习
type FeedbackLog struct {
    ID         int64
    TenantID   int64
    UserID     int64
    Input      string  // 用户原始说法
    SystemGuess string // 系统猜测
    UserChoice string  // 用户选择
    Correct    bool    // 系统猜测是否正确
    CreatedAt  time.Time
}

func (e *CorrectionEngineImpl) RecordFeedback(ctx context.Context, log FeedbackLog) error {
    // 写入 feedback 表
    if err := e.repo.SaveFeedback(ctx, log); err != nil {
        return err
    }

    // 如果用户纠正了系统猜测，更新字典（增加别名或提高权重）
    if !log.Correct && log.UserChoice != "" {
        _ = e.dictCenter.UpsertAlias(ctx, AliasUpsert{
            TenantID: fmt.Sprint(log.TenantID),
            Word:     log.UserChoice,
            Alias:    log.Input,
            Type:     "feedback",
            Weight:   3.0, // 初始较低权重
        })
    }

    return nil
}
```
