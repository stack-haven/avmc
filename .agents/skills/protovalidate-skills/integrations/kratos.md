## Middleware Implementation

### Basic Validator Middleware

```go
// pkg/middleware/validate.go
package middleware

import (
	"context"
	stderrors "errors"

	"buf.build/go/protovalidate"
	"github.com/go-kratos/kratos/v2/middleware"
	"google.golang.org/protobuf/proto"

	v1 "your-project/api/v1"
)

// validator is a singleton instance (thread-safe)
var validator protovalidate.Validator

func init() {
	var err error
	validator, err = protovalidate.New()
	if err != nil {
		panic(err)
	}
}

// Validator returns a protovalidate validation middleware
// It validates proto request messages against defined rules
func Validator() middleware.Middleware {
	return func(handler middleware.Handler) middleware.Handler {
		return func(ctx context.Context, req interface{}) (reply interface{}, err error) {
			// Check if request is a proto message
			if msg, ok := req.(proto.Message); ok {
				if err := validator.Validate(msg); err != nil {
					// Try to extract custom error message
					var valErr *protovalidate.ValidationError
					if stderrors.As(err, &valErr) {
						violations := valErr.Violations
						if len(violations) > 0 && violations[0].Proto.GetMessage() != "" {
							return nil, v1.ErrorBadRequest(violations[0].Proto.GetMessage())
						}
					}
					return nil, v1.ErrorBadRequest(err.Error())
				}
			}
			return handler(ctx, req)
		}
	}
}
```

package api.v1;

enum ErrorReason {

## Usage Example

### Proto Definition

```protobuf
// api/v1/user.proto
syntax = "proto3";

package api.v1;

import "buf/validate/validate.proto";
import "google/api/annotations.proto";

service UserService {
  rpc CreateUser (CreateUserRequest) returns (CreateUserReply) {
    option (google.api.http) = {
      post: "/api/v1/users"
      body: "*"
    };
  }
}

message CreateUserRequest {
  string username = 1 [(buf.validate.field).cel = {
    id: "username_format",
    message: "用户名必须为3-32个字符，只能包含字母、数字和下划线",
    expression: "this.matches('^[a-zA-Z0-9_]{3,32}$')"
  }];

  string password = 2 [(buf.validate.field).cel = {
    id: "password_length",
    message: "密码必须为6-32个字符",
    expression: "size(this) >= 6 && size(this) <= 32"
  }];

  string mobile = 3 [(buf.validate.field).cel = {
    id: "mobile_format",
    message: "手机号格式错误",
    expression: "this.matches('^1[3-9][0-9]{9}$')"
  }];
}
```

### Service Handler

```go
// internal/service/user.go
package service

import (
	"context"

	v1 "your-project/api/v1"
)

type UserService struct {
	v1.UnimplementedUserServiceServer

	uc *biz.UserUsecase
}

func NewUserService(uc *biz.UserUsecase) *UserService {
	return &UserService{uc: uc}
}

func (s *UserService) CreateUser(ctx context.Context, req *v1.CreateUserRequest) (*v1.CreateUserReply, error) {
	// Validation is already done by middleware
	// No need to validate here

	user, err := s.uc.Create(ctx, req)
	if err != nil {
		return nil, err
	}

	return &v1.CreateUserReply{
		Id: user.ID,
	}, nil
}
```

## Testing

### Unit Test for Validation

```go
// pkg/middleware/validate_test.go
package middleware

import (
	"testing"

	"buf.build/go/protovalidate"
	v1 "your-project/api/v1"
)

func TestValidation(t *testing.T) {
	validator, err := protovalidate.New()
	if err != nil {
		t.Fatalf("Failed to create validator: %v", err)
	}

	tests := []struct {
		name    string
		req     *v1.CreateUserRequest
		wantErr bool
	}{
		{
			name: "valid request",
			req: &v1.CreateUserRequest{
				Username: "testuser",
				Password: "password123",
				Mobile:   "13800138000",
			},
			wantErr: false,
		},
		{
			name: "invalid username",
			req: &v1.CreateUserRequest{
				Username: "ab", // too short
				Password: "password123",
				Mobile:   "13800138000",
			},
			wantErr: true,
		},
		{
			name: "invalid mobile",
			req: &v1.CreateUserRequest{
				Username: "testuser",
				Password: "password123",
				Mobile:   "12345", // invalid format
			},
			wantErr: true,
		},
	}

	for _, tt := range tests {
		t.Run(tt.name, func(t *testing.T) {
			err := validator.Validate(tt.req)
			if (err != nil) != tt.wantErr {
				t.Errorf("Validate() error = %v, wantErr %v", err, tt.wantErr)
			}
		})
	}
}
```

## Troubleshooting

### Common Errors

1. **"protovalidate module not excluded"**
   - Add `buf.build/bufbuild/protovalidate` to `buf.gen.yaml` managed disable list

2. **"undefined: protovalidate.Validator"**
   - Run `go get buf.build/go/protovalidate`

3. **"validation not working"**
   - Ensure middleware is added to server
   - Check that request is a proto.Message

4. **CEL regex error with `\d`**
   - Use `[0-9]` instead of `\d` in CEL expressions
